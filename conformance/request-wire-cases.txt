# Adversarial wire conformance corpus for host/http_parser.c
#
# Run by tests/test_http_conformance.c (the Makefile passes this file's path).
# Every vector is the EXACT byte sequence a client puts on the socket, spelled
# with escapes so no literal CR, LF or NUL ever appears in this file and no
# editor or git CRLF rule can silently rewrite a vector.
#
# Each case is three lines, in order, blank-line separated:
#   name:   <the rule or property this vector exercises>
#   expect: accept cl=<n> | accept nocl | reject <REASON> | incomplete
#   wire:   <escaped request bytes>
# REASON is one of BAD_REQUEST, SMUGGLING, HEADERS_TOO_LARGE, VERSION.
# Escapes in wire: \r \n \t \0 \\ and \xHH. Everything else is literal.
# A leading space or tab in a vector is written \x20 / \t so the "wire: "
# separator stays unambiguous.
#
# These are PARSER verdicts, not server verdicts: the parser validates framing,
# the request line and the header grammar only. It does not check the method,
# the Host value, Content-Type, Accept or Origin, so several vectors below the
# parser ACCEPTS are ones the server layer would still refuse (a GET, a bad
# Host). That boundary is the point of testing the parser in isolation.

# ============================================================ acceptance
name: a well-formed POST is accepted
expect: accept cl=2
wire: POST /mcp HTTP/1.1\r\nHost: 127.0.0.1:8080\r\nContent-Type: application/json\r\nAccept: application/json, text/event-stream\r\nContent-Length: 2\r\n\r\n{}

name: a minimal Host-only POST with an empty body is accepted
expect: accept cl=0
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length: 0\r\n\r\n

name: an absent Content-Length parses; framing is simply "none"
expect: accept nocl
wire: POST /mcp HTTP/1.1\r\nHost: x\r\n\r\n

name: a GET is accepted by the PARSER; the 405 is the server layer's
expect: accept nocl
wire: GET /mcp HTTP/1.1\r\nHost: x\r\n\r\n

name: a Content-Length with leading zeroes is 1*DIGIT and is accepted
expect: accept cl=7
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length: 007\r\n\r\n

name: Content-Length wrapped in HTAB OWS is trimmed and accepted
expect: accept cl=2
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length:\t2\t\r\n\r\n{}

name: Content-Length-Extra is not Content-Length, so no framing is set
expect: accept nocl
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length-Extra: 9\r\n\r\n

name: Transfer-Encoding-Foo is not Transfer-Encoding and does not desync
expect: accept cl=2
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nTransfer-Encoding-Foo: chunked\r\nContent-Length: 2\r\n\r\n{}

name: a repeated Mcp-Param-* is forwarded-and-ignored, not refused
expect: accept cl=2
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nMcp-Param-Foo: a\r\nMcp-Param-Foo: b\r\nContent-Length: 2\r\n\r\n{}

name: colons inside a header value belong to the value, not a second split
expect: accept nocl
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nX-Trace: a:b:c\r\n\r\n

name: a HTAB inside a header value is legal and kept
expect: accept nocl
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nX-Note:\tv\ta\tb\r\n\r\n

name: high bytes in a value are not controls and are accepted
expect: accept nocl
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nX-Bin: \xff\xfe\x80\r\n\r\n

name: the first of two pipelined requests completes with its own length
expect: accept cl=2
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\n\r\n{}POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\n\r\n{}

# ================================================ the smuggling class (TE + CL)
name: Transfer-Encoding then Content-Length is smuggling
expect: reject SMUGGLING
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nTransfer-Encoding: chunked\r\nContent-Length: 2\r\n\r\n{}

name: Content-Length then Transfer-Encoding is smuggling in either order
expect: reject SMUGGLING
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\nTransfer-Encoding: identity\r\n\r\n{}

name: case-obfuscated Transfer-Encoding with Content-Length is still smuggling
expect: reject SMUGGLING
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nTrAnSfEr-EnCoDiNg: chunked\r\nContent-Length: 2\r\n\r\n{}

name: a HTAB-led Transfer-Encoding value with Content-Length is smuggling
expect: reject SMUGGLING
wire: POST /mcp HTTP/1.1\r\nHost: x\r\nTransfer-Encoding:\tchunked\r\nContent-Length: 2\r\n\r\n{}

name: a whitespace-only Transfer-Encoding value is still present
expect: reject BAD_REQUEST

name: Transfer-Encoding as a chunked,chunked list is refused
